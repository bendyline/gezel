cmake_minimum_required(VERSION 3.22)
project(gezel_mobile LANGUAGES CXX)

get_filename_component(GEZEL_REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(GEZEL_JNI_LIBS "${CMAKE_CURRENT_LIST_DIR}/../../../android/app/src/main/jniLibs")
set(GEZEL_LLAMA_LIBRARY "${GEZEL_JNI_LIBS}/${ANDROID_ABI}/libgezel-llama.so")
if(NOT EXISTS "${GEZEL_LLAMA_LIBRARY}")
  message(FATAL_ERROR "Missing mobile llama libraries. Run native/mobile/build-llama.py android, then packages/mobile/scripts/sync-native.mjs.")
endif()
add_library(gezel_llama SHARED IMPORTED)
set_target_properties(gezel_llama PROPERTIES IMPORTED_LOCATION "${GEZEL_LLAMA_LIBRARY}")
add_library(gezel_mobile SHARED gezel_mobile_jni.cpp)
target_compile_features(gezel_mobile PRIVATE cxx_std_17)
target_include_directories(gezel_mobile PRIVATE "${GEZEL_REPO_ROOT}/native/mobile")
target_link_libraries(gezel_mobile PRIVATE gezel_llama)
target_link_options(gezel_mobile PRIVATE "-Wl,-z,max-page-size=16384")
add_library(gezel_speech SHARED IMPORTED)
set_target_properties(gezel_speech PROPERTIES IMPORTED_LOCATION "${GEZEL_JNI_LIBS}/${ANDROID_ABI}/libGezelSpeech.so")
target_include_directories(gezel_mobile PRIVATE "${GEZEL_REPO_ROOT}/native/mobile/speech")
target_link_libraries(gezel_mobile PRIVATE gezel_speech)
