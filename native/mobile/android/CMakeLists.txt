cmake_minimum_required(VERSION 3.22)
project(gezel-llama-jni LANGUAGES CXX)

if(NOT ANDROID)
    message(FATAL_ERROR "The Gezel JNI adapter requires an Android toolchain")
endif()
if(NOT EXISTS "${GEZEL_JNI_LIBS}/${ANDROID_ABI}/libgezel-llama.so")
    message(FATAL_ERROR "GEZEL_JNI_LIBS must contain the prebuilt llama runtime")
endif()
if(NOT EXISTS "${GEZEL_LLAMA_INCLUDE}/gezel_llama.h")
    message(FATAL_ERROR "GEZEL_LLAMA_INCLUDE must contain the matching Gezel C ABI header")
endif()
add_library(gezel_llama SHARED IMPORTED)
set_target_properties(gezel_llama PROPERTIES
    IMPORTED_LOCATION "${GEZEL_JNI_LIBS}/${ANDROID_ABI}/libgezel-llama.so")
add_library(gezel_llama_jni SHARED llama_jni.cpp)
target_compile_features(gezel_llama_jni PRIVATE cxx_std_17)
target_include_directories(gezel_llama_jni PRIVATE "${GEZEL_LLAMA_INCLUDE}")
target_link_libraries(gezel_llama_jni PRIVATE gezel_llama)
target_link_options(gezel_llama_jni PRIVATE "-Wl,-z,max-page-size=16384")
