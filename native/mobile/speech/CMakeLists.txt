cmake_minimum_required(VERSION 3.24)
project(gezel-mobile-speech LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
add_subdirectory("${GEZEL_WHISPER_SOURCE}" whisper)
add_library(GezelSpeech SHARED gezel_speech.cpp)
target_compile_features(GezelSpeech PRIVATE cxx_std_17)
target_include_directories(GezelSpeech PRIVATE "${GEZEL_SHERPA_HEADERS}")
target_link_libraries(GezelSpeech PRIVATE whisper "${GEZEL_SHERPA_LIBRARY}" "${GEZEL_ONNX_LIBRARY}")
if(APPLE)
  target_link_libraries(GezelSpeech PRIVATE "-framework Foundation" "-framework Accelerate" "-framework CoreML")
  set_target_properties(GezelSpeech PROPERTIES FRAMEWORK TRUE FRAMEWORK_VERSION A
    MACOSX_FRAMEWORK_IDENTIFIER com.bendyline.gezel.speech PUBLIC_HEADER gezel_speech.h)
elseif(ANDROID)
  target_link_options(GezelSpeech PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
endif()
